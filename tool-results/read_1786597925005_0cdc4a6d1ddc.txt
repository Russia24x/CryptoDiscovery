   106→/** Format a USD amount using K/M/B/T suffixes. e.g. $1.2B, $340M, $0.034 */
   107→function fmtUsd(n: number | null | undefined): string {
   108→  if (n == null || Number.isNaN(n)) return "—";
   109→  const abs = Math.abs(n);
   110→  if (abs >= 1e12) return `$${(n / 1e12).toFixed(2)}T`;
   111→  if (abs >= 1e9) return `$${(n / 1e9).toFixed(2)}B`;
   112→  if (abs >= 1e6) return `$${(n / 1e6).toFixed(1)}M`;
   113→  if (abs >= 1e3) return `$${(n / 1e3).toFixed(1)}K`;
   114→  if (abs >= 1) return `$${n.toFixed(2)}`;
   115→  return `$${n.toFixed(4)}`;
   116→}
   117→
   118→/** Format a percentage. Optionally prefix '+' on positive values. */
   119→function fmtPct(n: number | null | undefined, withSign = false): string {
   120→  if (n == null || Number.isNaN(n)) return "—";
   121→  const sign = withSign && n > 0 ? "+" : "";
   122→  return `${sign}${n.toFixed(1)}%`;
   123→}

   125→/** Tailwind text colour class for a 0–100 score. */
   126→function scoreTextClass(score: number): string {
   127→  if (score >= 70) return "text-emerald-400";
   128→  if (score >= 55) return "text-lime-400";
   129→  if (score >= 50) return "text-amber-400";
   130→  if (score >= 40) return "text-orange-400";
   131→  return "text-rose-400";
   132→}
   133→
   134→/** Tailwind bg/border classes for the action label badge (mirrors page.tsx). */
   135→function actionBadgeClass(actionLabel: string): string {
   136→  const a = (actionLabel || "").toLowerCase();
   137→  if (a.includes("high conviction")) return "bg-emerald-500/15 text-emerald-400 border-emerald-500/30";
   138→  if (a.includes("core")) return "bg-lime-500/15 text-lime-400 border-lime-500/30";
   139→  if (a.includes("small")) return "bg-amber-500/15 text-amber-400 border-amber-500/30";
   140→  if (a.includes("deep research") || a.includes("research")) return "bg-sky-500/15 text-sky-400 border-sky-500/30";
   141→  if (a.includes("watch")) return "bg-slate-500/15 text-slate-300 border-slate-500/30";
   142→  return "bg-rose-500/15 text-rose-400 border-rose-500/30";
   143→}

   145→/** Inline hex colour for an axis bar (0–10 scale → 0–100 normalised). */
   146→function axisBarColor(score: number): string {
   147→  const ratio = score / 10;
   148→  if (ratio >= 0.85) return "#10b981"; // emerald
   149→  if (ratio >= 0.70) return "#84cc16"; // lime
   150→  if (ratio >= 0.55) return "#f59e0b"; // amber
   151→  if (ratio >= 0.40) return "#f97316"; // orange
   152→  return "#f43f5e"; // rose
   153→}

   159→export function CoinExplorerView({
   160→  initialGeckoId,
   161→  onClearInitial,
   162→  onReport,
   163→}: CoinExplorerViewProps) {
   164→  const { t, lang } = useLanguage();
   165→
   166→  // Translate-with-fallback: returns `fallback` if the i18n key is missing.
   167→  // The LanguageProvider returns the raw key path when no translation is
   168→  // found, so we detect that case and substitute the English fallback.
   169→  const tf = React.useCallback(
   170→    (key: string, fallback: string, vars?: Record<string, string | number>): string => {
   171→      const val = t(key, vars);
   172→      if (typeof val !== "string") return fallback;
   173→      return val === key ? fallback : val;
   174→    },
   175→    [t],
   176→  );
   177→
   178→  // ----- search state -----
   179→  const [query, setQuery] = React.useState("");
   180→  const [debouncedQuery, setDebouncedQuery] = React.useState("");
   181→  const [results, setResults] = React.useState<CoinSearchResult[]>([]);
   182→  const [searching, setSearching] = React.useState(false);
   183→  const [showResults, setShowResults] = React.useState(false);
   184→  const [searchError, setSearchError] = React.useState<string | null>(null);
   185→  const searchInputRef = React.useRef<HTMLInputElement | null>(null);
   186→  const resultsContainerRef = React.useRef<HTMLDivElement | null>(null);
   187→
   188→  // ----- selection / persona / analysis state -----
   189→  const [selected, setSelected] = React.useState<CoinSearchResult | null>(null);
   190→  const [persona, setPersona] = React.useState<Persona>("investor");
   191→  const [analyzing, setAnalyzing] = React.useState(false);
   192→  const [phaseIdx, setPhaseIdx] = React.useState(0);
   193→  const [report, setReport] = React.useState<FullReport | null>(null);
   194→  const [error, setError] = React.useState<string | null>(null);
   195→
   196→  const initialConsumedRef = React.useRef(false);
   197→  const analyzeAbortRef = React.useRef<AbortController | null>(null);
   198→
   199→  // ----------------------------------------------------------------------- //
   200→  //  Debounced search query (300 ms)
   201→  // ----------------------------------------------------------------------- //
   202→  React.useEffect(() => {
   203→    const handle = window.setTimeout(() => setDebouncedQuery(query.trim()), 300);
   204→    return () => window.clearTimeout(handle);
   205→  }, [query]);
   206→
   207→  // ----------------------------------------------------------------------- //
   208→  //  Trigger search whenever debounced query changes
   209→  // ----------------------------------------------------------------------- //
   210→  React.useEffect(() => {
   211→    const q = debouncedQuery;
   212→    if (q.length < 1) {
   213→      setResults([]);
   214→      setSearching(false);
   215→      setSearchError(null);
   216→      return;
   217→    }
   218→    let cancelled = false;
   219→    setSearching(true);
   220→    setSearchError(null);
   221→
   222→    (async () => {
   223→      try {
   224→        const res = await fetch(`/api/scanner/search?q=${encodeURIComponent(q)}`, {
   225→          method: "GET",
   226→        });
   227→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   228→        const data = (await res.json()) as {
   229→          results?: CoinSearchResult[];
   230→          error?: string;
   231→        };
   232→        if (cancelled) return;
   233→        setResults(Array.isArray(data.results) ? data.results : []);
   234→      } catch (e) {
   235→        if (cancelled) return;
   236→        setSearchError(e instanceof Error ? e.message : "Search failed");
   237→        setResults([]);
   238→      } finally {
   239→        if (!cancelled) setSearching(false);
   240→      }
   241→    })();
   242→
   243→    return () => {
   244→      cancelled = true;
   245→    };
   246→  }, [debouncedQuery]);
   247→
   248→  // ----------------------------------------------------------------------- //
   249→  //  Cycle the loading phase message while analysis is in flight
   250→  // ----------------------------------------------------------------------- //
   251→  React.useEffect(() => {
   252→    if (!analyzing) return;
   253→    setPhaseIdx(0);
   254→    const id = window.setInterval(() => {
   255→      setPhaseIdx((i) => (i + 1) % ANALYZE_PHASE_MESSAGES.length);
   256→    }, 2200);
   257→    return () => window.clearInterval(id);
   258→  }, [analyzing]);
   259→
   260→  // ----------------------------------------------------------------------- //
   261→  //  Auto-select a coin when initialGeckoId is provided
   262→  // ----------------------------------------------------------------------- //
   263→  React.useEffect(() => {
   264→    if (initialConsumedRef.current) return;
   265→    if (!initialGeckoId) return;
   266→    initialConsumedRef.current = true;
   267→
   268→    let cancelled = false;
   269→    const geckoId = initialGeckoId;
   270→
   271→    (async () => {
   272→      try {
   273→        // Search using the gecko_id as the query — CoinGecko's search returns
   274→        // the exact-match coin at the top of the list in the vast majority of
   275→        // cases (e.g. "bitcoin" → "Bitcoin" #1).
   276→        const res = await fetch(
   277→          `/api/scanner/search?q=${encodeURIComponent(geckoId)}`,
   278→          { method: "GET" },
   279→        );
   280→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   281→        const data = (await res.json()) as { results?: CoinSearchResult[] };
   282→        if (cancelled) return;
   283→        const list = Array.isArray(data.results) ? data.results : [];
   284→        const match =
   285→          list.find((r) => r.id === geckoId) || list[0] || null;
   286→        if (match) {
   287→          setSelected(match);
   288→        } else {
   289→          // Fallback: synthesise a minimal CoinSearchResult so the persona
   290→          // panel can still be shown.
   291→          setSelected({
   292→            id: geckoId,
   293→            name: geckoId,
   294→            symbol: geckoId.toUpperCase().slice(0, 6),
   295→            market_cap_rank: null,
   296→            thumb: null,
   297→            large: null,
   298→            api_symbol: geckoId,
   299→          });
   300→        }
   301→      } catch {
   302→        if (cancelled) return;
   303→        // Last-resort fallback so the user can still trigger an analysis.
   304→        setSelected({
   305→          id: geckoId,
   306→          name: geckoId,
   307→          symbol: geckoId.toUpperCase().slice(0, 6),
   308→          market_cap_rank: null,
   309→          thumb: null,
   310→          large: null,
   311→          api_symbol: geckoId,
   312→        });
   313→      } finally {
   314→        if (!cancelled && onClearInitial) onClearInitial();
   315→      }
   316→    })();
   317→
   318→    return () => {
   319→      cancelled = true;
   320→    };
   321→  }, [initialGeckoId, onClearInitial]);
   322→
   323→  // ----------------------------------------------------------------------- //
   324→  //  Close search results dropdown on outside click / Escape
   325→  // ----------------------------------------------------------------------- //
   326→  React.useEffect(() => {
   327→    if (!showResults) return;
   328→    const onPointerDown = (e: MouseEvent) => {
   329→      if (
   330→        resultsContainerRef.current &&
   331→        !resultsContainerRef.current.contains(e.target as Node)
   332→      ) {
   333→        setShowResults(false);
   334→      }
   335→    };
   336→    const onKey = (e: KeyboardEvent) => {
   337→      if (e.key === "Escape") setShowResults(false);
   338→    };
   339→    document.addEventListener("mousedown", onPointerDown);
   340→    document.addEventListener("keydown", onKey);
   341→    return () => {
   342→      document.removeEventListener("mousedown", onPointerDown);
   343→      document.removeEventListener("keydown", onKey);
   344→    };
   345→  }, [showResults]);
   346→
   347→  // ----------------------------------------------------------------------- //
   348→  //  Cleanup: abort any in-flight analysis on unmount
   349→  // ----------------------------------------------------------------------- //
   350→  React.useEffect(() => {
   351→    return () => {
   352→      analyzeAbortRef.current?.abort();
   353→    };
   354→  }, []);
   355→
   356→  // ----------------------------------------------------------------------- //
   357→  //  Handlers
   358→  // ----------------------------------------------------------------------- //
   359→
   360→  const handleSelectCoin = React.useCallback((coin: CoinSearchResult) => {
   361→    setSelected(coin);
   362→    setShowResults(false);
   363→    setQuery("");
   364→    setResults([]);
   365→    setReport(null);
   366→    setError(null);
   367→    searchInputRef.current?.blur();
   368→  }, []);
   369→
   370→  const handleClearCoin = React.useCallback(() => {
   371→    setSelected(null);
   372→    setReport(null);
   373→    setError(null);
   374→    setPersona("investor");
   375→    analyzeAbortRef.current?.abort();
   376→    setAnalyzing(false);
   377→  }, []);
   378→
   379→  const runAnalyze = React.useCallback(async () => {
   380→    if (!selected) return;
   381→    // Abort any previous in-flight request.
   382→    analyzeAbortRef.current?.abort();
   383→    const ctrl = new AbortController();
   384→    analyzeAbortRef.current = ctrl;
   385→
   386→    setAnalyzing(true);
   387→    setError(null);
   388→    setReport(null);
   389→
   390→    try {
   391→      const res = await fetch("/api/scanner/analyze", {
   392→        method: "POST",
   393→        headers: { "Content-Type": "application/json" },
   394→        body: JSON.stringify({
   395→          gecko_id: selected.id,
   396→          persona,
   397→          lang,
   398→        }),
   399→        signal: ctrl.signal,
   400→      });
   401→
   402→      if (!res.ok) {
   403→        let detail = `HTTP ${res.status}`;
   404→        try {
   405→          const data = await res.json();
   406→          if (data?.error) detail = data.error;
   407→          if (data?.detail) detail = `${detail} — ${data.detail}`;
   408→        } catch {
   409→          /* ignore JSON parse failure */
   410→        }
   411→        throw new Error(detail);
   412→      }
   413→
   414→      const data = (await res.json()) as FullReport | { error?: string };
   415→      if (data && typeof data === "object" && "error" in data && data.error) {
   416→        throw new Error(data.error);
   417→      }
   418→      setReport(data as FullReport);
   419→    } catch (e) {
   420→      if ((e as Error)?.name === "AbortError") return; // user cancelled
   421→      setError(e instanceof Error ? e.message : "Analysis failed");
   422→    } finally {
   423→      if (analyzeAbortRef.current === ctrl) {
   424→        analyzeAbortRef.current = null;
   425→        setAnalyzing(false);
   426→      }
   427→    }
   428→  }, [selected, persona, lang]);
   429→
   430→  const handleViewFullReport = React.useCallback(() => {
   431→    if (report && onReport) onReport(report);
   432→  }, [report, onReport]);
   433→
   434→  // ----------------------------------------------------------------------- //
   435→  //  Render
   436→  // ----------------------------------------------------------------------- //
   437→
   438→  return (
   439→    <div className="flex flex-col gap-6 w-full">
   440→      {/* ----------------------------------------------------------------- */}
   441→      {/*  Search bar (always visible at the top)                            */}
   442→      {/* ----------------------------------------------------------------- */}
   443→      <Card className="border-border/60 bg-card/40 backdrop-blur-sm shadow-lg shadow-black/20">
   444→        <CardContent className="pt-0">
   445→          <div className="relative" ref={resultsContainerRef}>
   446→            <div className="flex items-center gap-2 rounded-lg border border-border/60 bg-input/30 px-3 py-2 transition-all focus-within:border-emerald-500/50 focus-within:ring-2 focus-within:ring-emerald-500/20">
   447→              <Search className="size-5 text-muted-foreground shrink-0" />
   448→              <input
   449→                ref={searchInputRef}
   450→                type="text"
   451→                value={query}
   452→                onChange={(e) => setQuery(e.target.value)}
   453→                onFocus={() => setShowResults(true)}
   454→                placeholder={
   455→                  tf("explorer.searchPlaceholder", "Search any coin by name or symbol…")
   456→                }
   457→                className="flex-1 bg-transparent text-base outline-none placeholder:text-muted-foreground text-foreground"
   458→                aria-label={tf("explorer.searchAria", "Search cryptocurrencies")}
   459→                autoComplete="off"
   460→                spellCheck={false}
   461→              />
   462→              {searching && (
   463→                <Loader2 className="size-4 animate-spin text-muted-foreground shrink-0" />
   464→              )}
   465→              {!searching && query && (
   466→                <button
   467→                  type="button"
   468→                  onClick={() => {
   469→                    setQuery("");
   470→                    setResults([]);
   471→                    searchInputRef.current?.focus();
   472→                  }}
   473→                  className="rounded-md p-1 text-muted-foreground hover:bg-accent hover:text-foreground transition-colors"
   474→                  aria-label={tf("explorer.clearSearch", "Clear search")}
   475→                >
   476→                  <X className="size-4" />
   477→                </button>
   478→              )}
   479→            </div>
   480→
   481→            {/* Search results dropdown */}
   482→            {showResults && debouncedQuery && (
   483→              <div className="absolute left-0 right-0 top-full z-30 mt-2">
   484→                <Card className="border-border/60 bg-popover/95 backdrop-blur-md shadow-2xl shadow-black/40 overflow-hidden">
   485→                  <CardContent className="p-0">
   486→                    {searchError && (
   487→                      <div className="p-3 text-sm text-rose-400 flex items-center gap-2">
   488→                        <AlertCircle className="size-4" />
   489→                        {tf("explorer.searchFailed", "Search failed")} — {searchError}
   490→                      </div>
   491→                    )}
   492→                    {!searchError && !searching && results.length === 0 && (
   493→                      <div className="flex flex-col items-center gap-2 py-8 px-4 text-center">
   494→                        <Frown className="size-6 text-muted-foreground" />
   495→                        <p className="text-sm text-muted-foreground">
   496→                          {tf("explorer.noResults", "No coins found for")}{" "}
   497→                          <span className="font-medium text-foreground">
   498→                            “{debouncedQuery}”
   499→                          </span>
   500→                        </p>
   501→                      </div>
   502→                    )}
   503→                    {results.length > 0 && (
   504→                      <ScrollArea className="max-h-96 overflow-y-auto">
   505→                        <ul className="py-1">
   506→                          {results.slice(0, 25).map((coin) => (
   507→                            <SearchResultRow
   508→                              key={coin.id}
   509→                              coin={coin}
   510→                              onSelect={handleSelectCoin}
   511→                              tf={tf}
   512→                            />
   513→                          ))}
   514→                        </ul>
   515→                      </ScrollArea>
   516→                    )}
   517→                    {searching && results.length === 0 && (
   518→                      <div className="p-3 space-y-2">
   519→                        {[0, 1, 2].map((i) => (
   520→                          <div key={i} className="flex items-center gap-3">
   521→                            <Skeleton className="size-8 rounded-full" />
   522→                            <div className="flex-1 space-y-1.5">
   523→                              <Skeleton className="h-3 w-1/3" />
   524→                              <Skeleton className="h-2 w-1/4" />
   525→                            </div>
   526→                          </div>
   527→                        ))}
   528→                      </div>
   529→                    )}
   530→                  </CardContent>
   531→                </Card>
   532→              </div>
   533→            )}
   534→          </div>
   535→        </CardContent>
   536→      </Card>
   537→
   538→      {/* ----------------------------------------------------------------- */}
   539→      {/*  Main panel — selected coin / persona / analyze / result           */}
   540→      {/* ----------------------------------------------------------------- */}
   541→      {!selected && !analyzing && !report && !error && <EmptyState tf={tf} />}
   542→
   543→      {selected && (
   544→        <div className="flex flex-col gap-6">
   545→          <SelectedCoinPanel
   546→            coin={selected}
   547→            persona={persona}
   548→            onPersonaChange={setPersona}
   549→            onAnalyze={runAnalyze}
   550→            onClear={handleClearCoin}
   551→            analyzing={analyzing}
   552→            tf={tf}
   553→          />
   554→
   555→          {analyzing && (
   556→            <AnalyzingCard phaseIdx={phaseIdx} tf={tf} />
   557→          )}
   558→
   559→          {!analyzing && error && (
   560→            <ErrorCard
   561→              message={error}
   562→              onRetry={runAnalyze}

   856→// --------------------------------------------------------------------------- //
   857→//  Analyzing card — full-card loading state with cycling phase messages
   858→// --------------------------------------------------------------------------- //
   859→
   860→function AnalyzingCard({
   861→  phaseIdx,
   862→  tf,
   863→}: {
   864→  phaseIdx: number;
   865→  tf: (key: string, fallback: string, vars?: Record<string, string | number>) => string;
   866→}) {
   867→  // Fake progress that asymptotically approaches 90% — gives the user a sense
   868→  // of forward motion without lying about the actual completion %.
   869→  const [progress, setProgress] = React.useState(2);
   870→  React.useEffect(() => {
   871→    const id = window.setInterval(() => {
   872→      setProgress((p) => {
   873→        if (p >= 92) return p;
   874→        // Decelerating curve: the closer we get to 92, the slower we move.
   875→        const remaining = 92 - p;
   876→        const step = Math.max(0.4, remaining * 0.06);
   877→        return Math.min(92, p + step);
   878→      });
   879→    }, 400);
   880→    return () => window.clearInterval(id);
   881→  }, []);
   882→
   883→  return (
   884→    <Card className="border-emerald-500/30 bg-card/40 backdrop-blur-sm shadow-xl shadow-emerald-500/10 overflow-hidden">
   885→      <div className="relative">
   886→        {/* Top accent bar */}
   887→        <div className="absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent via-emerald-400/80 to-transparent" />
   888→        <CardContent className="py-12 flex flex-col items-center text-center gap-6">
   889→          {/* Animated radar-like spinner */}
   890→          <div className="relative size-20">
   891→            <div className="absolute inset-0 rounded-full border-2 border-emerald-500/20" />
   892→            <div className="absolute inset-0 rounded-full border-t-2 border-emerald-400 animate-spin" style={{ animationDuration: "1.4s" }} />
   893→            <div className="absolute inset-2 rounded-full border-2 border-teal-500/10" />
   894→            <div className="absolute inset-2 rounded-full border-b-2 border-teal-400 animate-spin" style={{ animationDuration: "2s", animationDirection: "reverse" }} />
   895→            <div className="absolute inset-0 flex items-center justify-center">
   896→              <Brain className="size-7 text-emerald-400" />
   897→            </div>
   898→          </div>
   899→
   900→          <div className="space-y-2">
   901→            <h3 className="text-lg font-semibold text-foreground flex items-center justify-center gap-2">
   902→              <Loader2 className="size-4 animate-spin text-emerald-400" />
   903→              {tf("explorer.analyzingTitle", "Running framework analysis")}
   904→            </h3>
   905→            <p
   906→              key={phaseIdx}
   907→              className="text-sm text-muted-foreground transition-opacity duration-300 max-w-md"
   908→            >
   909→              {ANALYZE_PHASE_MESSAGES[phaseIdx]}
   910→            </p>
   911→          </div>
   912→
   913→          {/* Phase dots */}
   914→          <div className="flex items-center gap-1.5">
   915→            {ANALYZE_PHASE_MESSAGES.map((_, i) => (
   916→              <span
   917→                key={i}
   918→                className={cn(
   919→                  "h-1.5 rounded-full transition-all duration-300",
   920→                  i === phaseIdx
   921→                    ? "w-6 bg-emerald-400"
   922→                    : i < phaseIdx
   923→                      ? "w-1.5 bg-emerald-500/50"
   924→                      : "w-1.5 bg-muted",
   925→                )}
   926→              />
   927→            ))}
   928→          </div>
   929→
   930→          {/* Progress bar */}
   931→          <div className="w-full max-w-md space-y-1.5">
   932→            <Progress
   933→              value={progress}
   934→              className="h-1.5 bg-muted/40 [&_[data-slot=progress-indicator]]:bg-gradient-to-r [&_[data-slot=progress-indicator]]:from-emerald-500 [&_[data-slot=progress-indicator]]:to-teal-400"

  1253→function KeyLinkIcon({
  1254→  href,
  1255→  icon,
  1256→  label,
  1257→}: {
  1258→  href: string | null | undefined;
  1259→  icon: React.ReactNode;
  1260→  label: string;
  1261→}) {
  1262→  if (!href) return null;
  1263→  let safeHref = href;
  1264→  try {
  1265→    // Ensure protocol-relative or bare URLs become absolute & https.
  1266→    if (!/^https?:\/\//i.test(href)) safeHref = `https://${href}`;
  1267→  } catch {
  1268→    /* ignore */
  1269→  }
  1270→  return (
  1271→    <Tooltip>
  1272→      <TooltipTrigger asChild>
  1273→        <a
  1274→          href={safeHref}
  1275→          target="_blank"
  1276→          rel="noopener noreferrer"
  1277→          aria-label={label}
  1278→          className="inline-flex size-8 items-center justify-center rounded-md border border-border/60 bg-input/20 text-muted-foreground transition-all hover:bg-accent/60 hover:text-foreground hover:border-border"
  1279→        >
  1280→          {icon}
  1281→        </a>
  1282→      </TooltipTrigger>
  1283→      <TooltipContent>{label}</TooltipContent>
  1284→    </Tooltip>
  1285→  );
  1286→}
