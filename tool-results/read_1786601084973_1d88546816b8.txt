   158→/** Translate-with-fallback helper, reusable by any sub-component. */
   159→function useTt() {
   160→  const { t } = useLanguage();
   161→  return React.useCallback(
   162→    (key: string, fallback: string, vars?: Record<string, string | number>): string => {
   163→      const v = t(key, vars);
   164→      return typeof v === "string" && v !== key ? v : fallback;
   165→    },
   166→    [t],
   167→  );
   168→}

   188→/** Format an ISO date string as a short relative time, e.g. "5m ago". */
   189→function timeAgo(iso: string | null): string {
   190→  if (!iso) return "—";
   191→  const date = new Date(iso);
   192→  if (isNaN(date.getTime())) return "—";
   193→  const diffSec = Math.max(0, Math.floor((Date.now() - date.getTime()) / 1000));
   194→  if (diffSec < 5) return "just now";
   195→  if (diffSec < 60) return `${diffSec}s ago`;
   196→  const min = Math.floor(diffSec / 60);
   197→  if (min < 60) return `${min}m ago`;
   198→  const hr = Math.floor(min / 60);
   199→  if (hr < 24) return `${hr}h ago`;
   200→  const day = Math.floor(hr / 24);
   201→  if (day < 7) return `${day}d ago`;
   202→  const wk = Math.floor(day / 7);
   203→  if (wk < 5) return `${wk}w ago`;
   204→  const mon = Math.floor(day / 30);
   205→  if (mon < 12) return `${mon}mo ago`;
   206→  const yr = Math.floor(day / 365);
   207→  return `${yr}y ago`;
   208→}
   209→
   210→/** Format an ISO date string as a clock time, e.g. "13:45" or "Aug 13, 13:45". */
   211→function timeClock(iso: string | null): string {
   212→  if (!iso) return "—";
   213→  const date = new Date(iso);
   214→  if (isNaN(date.getTime())) return "—";
   215→  const now = new Date();
   216→  const sameDay =
   217→    date.getFullYear() === now.getFullYear() &&
   218→    date.getMonth() === now.getMonth() &&
   219→    date.getDate() === now.getDate();
   220→  const time = date.toLocaleTimeString(undefined, {
   221→    hour: "2-digit",
   222→    minute: "2-digit",
   223→  });
   224→  if (sameDay) return time;
   225→  const dateStr = date.toLocaleDateString(undefined, {
   226→    month: "short",
   227→    day: "numeric",
   228→  });
   229→  return `${dateStr}, ${time}`;
   230→}
   231→
   232→/** Seconds since an ISO date. Returns 0 when invalid. */
   233→function secondsSince(iso: string | null): number {
   234→  if (!iso) return 0;
   235→  const date = new Date(iso);
   236→  if (isNaN(date.getTime())) return 0;
   237→  return Math.max(0, Math.floor((Date.now() - date.getTime()) / 1000));
   238→}

  1085→  // ----- Tab state -----

  1099→  const [newsFaRefreshing, setNewsFaRefreshing] = React.useState(false);
  1100→  const [newsFaError, setNewsFaError] = React.useState<string | null>(null);
  1101→  const [categoryFilter, setCategoryFilter] = React.useState<string>("all");
  1102→  const [searchFa, setSearchFa] = React.useState("");
  1103→
  1104→  // ----- Telegram state -----
  1105→  const [telegram, setTelegram] = React.useState<TelegramResponse | null>(null);
  1106→  const [tgLoading, setTgLoading] = React.useState(false);
  1107→  const [tgRefreshing, setTgRefreshing] = React.useState(false);
  1108→  const [tgError, setTgError] = React.useState<string | null>(null);

  1111→  // ----- Sources state -----

  1113→

  1120→

  1129→      const data = (await res.json()) as NewsResponse;
  1130→      if (!data || !Array.isArray(data.articles)) {
  1131→        throw new Error("Invalid response shape");
  1132→      }
  1133→      setNews(data);
  1134→    } catch (e) {
  1135→      setNewsError(e instanceof Error ? e.message : "Failed to load news");
  1136→    } finally {
  1137→      setNewsLoading(false);
  1138→      setNewsRefreshing(false);
  1139→    }
  1140→  }, []);
  1141→
  1142→  const fetchNewsFa = React.useCallback(async (refresh: boolean) => {
  1143→    if (refresh) setNewsFaRefreshing(true);
  1144→    else setNewsFaLoading(true);
  1145→    setNewsFaError(null);
  1146→    try {
  1147→      const res = await fetch("/api/scanner/news/fa?limit=40");
  1148→      if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
  1149→      const data = (await res.json()) as PersianNewsResponse;
  1150→      if (!data || !Array.isArray(data.articles)) {
  1151→        throw new Error("Invalid response shape");
  1152→      }
  1153→      setNewsFa(data);
  1154→    } catch (e) {
  1155→      setNewsFaError(e instanceof Error ? e.message : "Failed to load Persian news");
  1156→    } finally {
  1157→      setNewsFaLoading(false);
  1158→      setNewsFaRefreshing(false);
  1159→    }
  1160→  }, []);
  1161→
  1162→  const fetchTelegram = React.useCallback(async (refresh: boolean) => {
  1163→    if (refresh) setTgRefreshing(true);
  1164→    else setTgLoading(true);
  1165→    setTgError(null);
  1166→    try {
  1167→      const res = await fetch(
  1168→        "/api/scanner/telegram?channel=Mastersharkcrypto&limit=20",

  1171→      const data = (await res.json()) as TelegramResponse;
  1172→      if (data?.error) throw new Error(data.error);
  1173→      if (!data || !Array.isArray(data.messages)) {
  1174→        throw new Error("Invalid response shape");
  1175→      }
  1176→      setTelegram(data);
  1177→    } catch (e) {
  1178→      setTgError(e instanceof Error ? e.message : "Failed to load telegram feed");
  1179→    } finally {
  1180→      setTgLoading(false);
  1181→      setTgRefreshing(false);
  1182→    }
  1183→  }, []);

  1185→  const fetchSources = React.useCallback(async () => {

  1187→      const res = await fetch("/api/scanner/sources");

  1189→      const data = (await res.json()) as SourcesStatus;
  1190→      if (data?.sources) setSources(data.sources);
  1191→    } catch {
  1192→      // Non-critical — silently ignore.
  1193→    }
  1194→  }, []);
  1195→
  1196→  // ----- Initial load (sources + active tab) -----
  1197→  React.useEffect(() => {
  1198→    fetchSources();
  1199→  }, [fetchSources]);
  1200→
  1201→  React.useEffect(() => {
  1202→    if (tab === "news" && !news && !newsLoading) fetchNews(false);
  1203→    else if (tab === "news_fa" && !newsFa && !newsFaLoading) fetchNewsFa(false);
  1204→    else if (tab === "telegram" && !telegram && !tgLoading) fetchTelegram(false);
  1205→  }, [tab, news, newsFa, telegram, newsLoading, newsFaLoading, tgLoading, fetchNews, fetchNewsFa, fetchTelegram]);
  1206→
  1207→  // ----- Auto-refresh telegram every 60s when toggle is on -----
  1208→  React.useEffect(() => {
  1209→    if (!autoRefresh) return;
  1210→    const id = setInterval(() => fetchTelegram(true), 60_000);
  1211→    return () => clearInterval(id);
  1212→  }, [autoRefresh, fetchTelegram]);
  1213→
  1214→  // ----- Derived: news sources list for filter chips -----
  1215→  const newsSources = React.useMemo(() => {
  1216→    const set = new Set<string>();
  1217→    news?.articles.forEach((a) => set.add(a.source));
  1218→    return Array.from(set);
  1219→  }, [news]);
  1220→
  1221→  // ----- Derived: filtered news articles (English) -----
  1222→  const filteredArticles = React.useMemo(() => {
  1223→    if (!news) return [];
  1224→    const q = search.trim().toLowerCase();
  1225→    return news.articles.filter((a) => {
  1226→      if (sourceFilter !== "all" && a.source !== sourceFilter) return false;
  1227→      if (q) {
  1228→        const hay = `${a.title} ${a.summary}`.toLowerCase();
  1229→        if (!hay.includes(q)) return false;
  1230→      }
  1231→      return true;
  1232→    });
  1233→  }, [news, sourceFilter, search]);
  1234→
  1235→  // ----- Derived: filtered Persian news articles -----
  1236→  const filteredFaArticles = React.useMemo(() => {
  1237→    if (!newsFa) return [];
  1238→    const q = searchFa.trim().toLowerCase();
  1239→    return newsFa.articles.filter((a) => {
  1240→      if (categoryFilter !== "all" && a.category !== categoryFilter) return false;
  1241→      if (q) {
  1242→        const hay = `${a.title} ${a.summary}`.toLowerCase();
  1243→        if (!hay.includes(q)) return false;
  1244→      }
  1245→      return true;
  1246→    });
  1247→  }, [newsFa, categoryFilter, searchFa]);
  1248→
  1249→  // ----- Derived: cache age for the active tab -----
  1250→  const activeFetchedAt =
  1251→    tab === "news" ? news?.fetched_at
  1252→    : tab === "news_fa" ? newsFa?.fetched_at
  1253→    : telegram?.fetched_at;
  1254→  // secondsSince recomputes on every render (tick state forces re-render every 1s).
  1255→  void secondsSince(activeFetchedAt ?? null);
  1256→
  1257→  // ----- Refresh handler for the current tab -----
  1258→  const handleRefresh = React.useCallback(() => {
  1259→    if (tab === "news") fetchNews(true);
  1260→    else if (tab === "news_fa") fetchNewsFa(true);
  1261→    else fetchTelegram(true);
  1262→  }, [tab, fetchNews, fetchNewsFa, fetchTelegram]);
  1263→
  1264→  const isRefreshing =
  1265→    tab === "news" ? newsRefreshing
  1266→    : tab === "news_fa" ? newsFaRefreshing
  1267→    : tgRefreshing;
  1268→  const isLoading =
  1269→    tab === "news" ? newsLoading
  1270→    : tab === "news_fa" ? newsFaLoading
  1271→    : tgLoading;
  1272→  const activeError =
  1273→    tab === "news" ? newsError
  1274→    : tab === "news_fa" ? newsFaError
  1275→    : tgError;
  1276→  const onRetry =
  1277→    tab === "news" ? () => fetchNews(false)
  1278→    : tab === "news_fa" ? () => fetchNewsFa(false)
  1279→    : () => fetchTelegram(false);
  1280→
  1281→  // ----------------------------------------------------------------------- //
  1282→  //  Render
  1283→  // ----------------------------------------------------------------------- //
  1284→
  1285→  return (
  1286→    <TooltipProvider>
  1287→      <div className="space-y-4">
  1288→        {/* ===== Header ===== */}
  1289→        <Card className="border-border/60 bg-card/40 backdrop-blur-sm">
  1290→          <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
  1291→            <div className="flex items-center gap-3">
  1292→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500/20 to-teal-500/20 text-emerald-600 dark:text-emerald-400">
  1293→                <Newspaper className="size-5" />
  1294→              </div>
  1295→              <div>
  1296→                <h2 className="text-lg font-semibold leading-tight">
  1297→                  {tt("news.title", "News & Signals")}
  1298→                </h2>
  1299→                <p className="text-xs text-muted-foreground">
  1300→                  {tt("news.subtitle", "Crypto news + Telegram channel, unified.")}
  1301→                </p>
  1302→              </div>
  1303→            </div>
  1304→            <div className="flex items-center gap-3">
  1305→              {activeFetchedAt && (
  1306→                <span className="hidden items-center gap-1 text-xs text-muted-foreground sm:flex">
  1307→                  <Clock className="size-3" />
  1308→                  {tt("news.updated", "Updated")} {timeAgo(activeFetchedAt)}
  1309→                </span>

  1311→              <Button
  1312→                size="sm"
  1313→                variant="outline"
  1314→                onClick={handleRefresh}
  1315→                disabled={isRefreshing || isLoading}

  1317→                <RefreshCw className={`size-3.5 ${isRefreshing ? "animate-spin" : ""}`} />
  1318→                <span className="hidden sm:inline">{tt("news.refresh", "Refresh")}</span>
