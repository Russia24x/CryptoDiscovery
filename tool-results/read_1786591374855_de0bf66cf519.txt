   260→function fmtUsd(v: number | null | undefined): string {
   261→  if (v == null || isNaN(v)) return "—";
   262→  const abs = Math.abs(v);
   263→  if (abs >= 1e9) return `$${(v / 1e9).toFixed(2)}B`;
   264→  if (abs >= 1e6) return `$${(v / 1e6).toFixed(1)}M`;
   265→  if (abs >= 1e3) return `$${(v / 1e3).toFixed(0)}K`;
   266→  return `$${v.toFixed(2)}`;
   267→}
   268→
   269→function fmtPct(v: number | null | undefined, digits = 1): string {
   270→  if (v == null || isNaN(v)) return "—";
   271→  return `${v.toFixed(digits)}%`;
   272→}

   274→function scoreColor(score: number, max = 100): string {
   275→  const ratio = score / max;
   276→  if (ratio >= 0.85) return "text-emerald-500";
   277→  if (ratio >= 0.70) return "text-lime-500";
   278→  if (ratio >= 0.55) return "text-amber-500";
   279→  if (ratio >= 0.40) return "text-orange-500";
   280→  return "text-rose-500";
   281→}
   282→
   283→function scoreBg(score: number, max = 100): string {
   284→  const ratio = score / max;
   285→  if (ratio >= 0.85) return "bg-emerald-500";
   286→  if (ratio >= 0.70) return "bg-lime-500";
   287→  if (ratio >= 0.55) return "bg-amber-500";
   288→  if (ratio >= 0.40) return "bg-orange-500";
   289→  return "bg-rose-500";
   290→}

   292→function actionBadge(action: string, t?: (key: string) => string): { label: string; cls: string } {
   293→  const a = action.toLowerCase();
   294→  let label = action;
   295→  if (t) {
   296→    if (a.includes("high conviction")) label = t("actions.highConviction");
   297→    else if (a.includes("core")) label = t("actions.coreCandidate");
   298→    else if (a.includes("small")) label = t("actions.smallPosition");
   299→    else if (a.includes("deep research") || a.includes("research")) label = a.includes("confidence") ? t("actions.deepResearchConfidence") : t("actions.deepResearch");
   300→    else if (a.includes("watch")) label = a.includes("low") ? t("actions.watchLowConfidence") : t("actions.watch");
   301→    else label = t("actions.ignore");
   302→  }
   303→  if (a.includes("high conviction")) return { label, cls: "bg-emerald-500/15 text-emerald-400 border-emerald-500/30" };
   304→  if (a.includes("core")) return { label, cls: "bg-lime-500/15 text-lime-400 border-lime-500/30" };
   305→  if (a.includes("small")) return { label, cls: "bg-amber-500/15 text-amber-400 border-amber-500/30" };
   306→  if (a.includes("deep research") || a.includes("research")) return { label, cls: "bg-sky-500/15 text-sky-400 border-sky-500/30" };
   307→  if (a.includes("watch")) return { label, cls: "bg-slate-500/15 text-slate-300 border-slate-500/30" };
   308→  return { label, cls: "bg-rose-500/15 text-rose-400 border-rose-500/30" };
   309→}
