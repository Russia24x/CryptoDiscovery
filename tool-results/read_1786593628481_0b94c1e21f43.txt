   105→/** `$1.27T` · `$1.2B` · `$340M` · `$12K` · `$0.034` · `—` for null. */
   106→function fmtUsd(v: number | null | undefined): string {
   107→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   108→  const abs = Math.abs(v);
   109→  const sign = v < 0 ? "-" : "";
   110→  if (abs >= 1e12) return `${sign}$${(abs / 1e12).toFixed(2)}T`;
   111→  if (abs >= 1e9) return `${sign}$${(abs / 1e9).toFixed(2)}B`;
   112→  if (abs >= 1e6) return `${sign}$${(abs / 1e6).toFixed(1)}M`;
   113→  if (abs >= 1e3) return `${sign}$${(abs / 1e3).toFixed(1)}K`;
   114→  if (abs >= 1) return `${sign}$${abs.toFixed(2)}`;
   115→  if (abs > 0) return `${sign}$${abs.toFixed(4)}`;
   116→  return "$0.00";
   117→}
   118→
   119→/** Like fmtUsd but tuned for token prices (more decimals for sub-$1 tokens). */
   120→function fmtPrice(v: number | null | undefined): string {
   121→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   122→  const abs = Math.abs(v);
   123→  if (abs >= 1e3) return `$${v.toLocaleString(undefined, { maximumFractionDigits: 0 })}`;
   124→  if (abs >= 1) return `$${v.toFixed(2)}`;
   125→  if (abs >= 0.01) return `$${v.toFixed(4)}`;
   126→  if (abs > 0) return `$${v.toExponential(3)}`;
   127→  return "$0.00";
   128→}
   129→
   130→/** `+1.2%` / `-2.1%` / `—`. */
   131→function fmtPct(v: number | null | undefined, digits = 2): string {
   132→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   133→  const s = v > 0 ? "+" : "";
   134→  return `${s}${v.toFixed(digits)}%`;
   135→}
   136→
   137→/** `18,404` with commas. */
   138→function fmtNum(v: number | null | undefined): string {
   139→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   140→  return v.toLocaleString();
   141→}
   142→
   143→/** Color class for a % change: emerald for positive, rose for negative. */
   144→function pctColor(v: number | null | undefined): string {
   145→  if (v == null || isNaN(v)) return "text-muted-foreground";
   146→  if (v > 0) return "text-emerald-500";
   147→  if (v < 0) return "text-rose-500";
   148→  return "text-muted-foreground";
   149→}

   204→function useMarketOverview() {
   205→  const [data, setData] = React.useState<MarketOverview | null>(null);
   206→  const [loading, setLoading] = React.useState(true);
   207→  const [refreshing, setRefreshing] = React.useState(false);
   208→  const [error, setError] = React.useState<string | null>(null);
   209→
   210→  const load = React.useCallback(async (isRefresh: boolean) => {
   211→    if (isRefresh) setRefreshing(true);
   212→    else setLoading(true);
   213→    setError(null);
   214→    try {
   215→      const ctrl = new AbortController();
   216→      const t = setTimeout(() => ctrl.abort(), 60_000);
   217→      const res = await fetch("/api/scanner/market/overview", {
   218→        signal: ctrl.signal,
   219→        cache: "no-store",
   220→      });
   221→      clearTimeout(t);
   222→      if (!res.ok) {
   223→        throw new Error(`HTTP ${res.status}`);
   224→      }
   225→      const json: MarketOverview = await res.json();
   226→      setData(json);
   227→    } catch (err) {
   228→      const msg = err instanceof Error ? err.message : String(err);
   229→      if (err instanceof DOMException && err.name === "AbortError") {
   230→        setError("Request timed out after 60s.");
   231→      } else {
   232→        setError(msg);
   233→      }
   234→    } finally {
   235→      setLoading(false);
   236→      setRefreshing(false);
   237→    }
   238→  }, []);
   239→
   240→  React.useEffect(() => {
   241→    load(false);
   242→  }, [load]);
   243→
   244→  return { data, loading, refreshing, error, refresh: () => load(true) };
   245→}

   247→/** Returns the age in seconds of the cached_at timestamp, re-ticking every 10s. */
   248→function useCacheAge(cachedAt: string | null | undefined): number | null {
   249→  const [now, setNow] = React.useState(() => Date.now());
   250→  React.useEffect(() => {
   251→    const id = setInterval(() => setNow(Date.now()), 10_000);
   252→    return () => clearInterval(id);
   253→  }, []);
   254→  if (!cachedAt) return null;
   255→  const t = Date.parse(cachedAt);
   256→  if (isNaN(t)) return null;
   257→  return Math.max(0, Math.floor((now - t) / 1000));
   258→}
   259→
   260→function formatAge(secs: number | null): string {
   261→  if (secs == null) return "";
   262→  if (secs < 60) return `${secs}s ago`;
   263→  const m = Math.floor(secs / 60);
   264→  const s = secs % 60;
   265→  if (m < 60) return `${m}m ${s}s ago`;
   266→  const h = Math.floor(m / 60);
   267→  return `${h}h ${m % 60}m ago`;
   268→}

   718→function SectorsChart({ sectors }: { sectors: SectorBreakdown[] }) {
   719→  const sorted = React.useMemo(
   720→    () => [...sectors].sort((a, b) => (b.total_market_cap || 0) - (a.total_market_cap || 0)),
   721→    [sectors],
   722→  );
   723→  const total = React.useMemo(
   724→    () => sorted.reduce((sum, s) => sum + (s.total_market_cap || 0), 0),
   725→    [sorted],
   726→  );
   727→  const [hovered, setHovered] = React.useState<number | null>(null);
   728→
   729→  if (sorted.length === 0) {
   730→    return (
   731→      <div className="flex h-32 items-center justify-center text-sm text-muted-foreground">
   732→        No sector data
   733→      </div>
   734→    );
   735→  }
   736→
   737→  return (
   738→    <div className="space-y-4">
   739→      {/* Proportional stacked bar */}
   740→      <div className="flex h-6 w-full overflow-hidden rounded-md border border-border/40">
   741→        {sorted.map((s, i) => {
   742→          const v = s.total_market_cap || 0;
   743→          const w = total > 0 ? (v / total) * 100 : 0;
   744→          if (w <= 0) return null;

   746→            <Tooltip key={s.sector}>
   747→              <TooltipTrigger asChild>
   748→                <div

   760→                <div className="text-xs">
   761→                  <div className="font-medium">{s.sector}</div>

   767→        })}

   769→

   773→          const v = s.total_market_cap || 0;
   774→          const w = total > 0 ? (v / total) * 100 : 0;

   785→                  className="size-2.5 shrink-0 rounded-sm"

   787→                />
   788→                <span className="truncate text-foreground">{s.sector}</span>
   789→              </div>
   790→              <div className="relative h-2.5 overflow-hidden rounded-full bg-muted/40">
   791→                <div
   792→                  className="h-full rounded-full transition-all"
   793→                  style={{
   794→                    width: `${w}%`,
   795→                    backgroundColor: color,
