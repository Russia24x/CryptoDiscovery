   176→/** Format an ISO date string as a short relative time, e.g. "5m ago". */
   177→function timeAgo(iso: string | null): string {
   178→  if (!iso) return "—";
   179→  const date = new Date(iso);
   180→  if (isNaN(date.getTime())) return "—";
   181→  const diffSec = Math.max(0, Math.floor((Date.now() - date.getTime()) / 1000));
   182→  if (diffSec < 5) return "just now";
   183→  if (diffSec < 60) return `${diffSec}s ago`;
   184→  const min = Math.floor(diffSec / 60);
   185→  if (min < 60) return `${min}m ago`;
   186→  const hr = Math.floor(min / 60);
   187→  if (hr < 24) return `${hr}h ago`;
   188→  const day = Math.floor(hr / 24);
   189→  if (day < 7) return `${day}d ago`;
   190→  const wk = Math.floor(day / 7);
   191→  if (wk < 5) return `${wk}w ago`;
   192→  const mon = Math.floor(day / 30);
   193→  if (mon < 12) return `${mon}mo ago`;
   194→  const yr = Math.floor(day / 365);
   195→  return `${yr}y ago`;
   196→}
   197→
   198→/** Format an ISO date string as a clock time, e.g. "13:45" or "Aug 13, 13:45". */
   199→function timeClock(iso: string | null): string {
   200→  if (!iso) return "—";
   201→  const date = new Date(iso);
   202→  if (isNaN(date.getTime())) return "—";
   203→  const now = new Date();
   204→  const sameDay =
   205→    date.getFullYear() === now.getFullYear() &&
   206→    date.getMonth() === now.getMonth() &&
   207→    date.getDate() === now.getDate();
   208→  const time = date.toLocaleTimeString(undefined, {
   209→    hour: "2-digit",
   210→    minute: "2-digit",
   211→  });
   212→  if (sameDay) return time;
   213→  const dateStr = date.toLocaleDateString(undefined, {
   214→    month: "short",
   215→    day: "numeric",
   216→  });
   217→  return `${dateStr}, ${time}`;
   218→}
   219→
   220→/** Seconds since an ISO date. Returns 0 when invalid. */
   221→function secondsSince(iso: string | null): number {
   222→  if (!iso) return 0;
   223→  const date = new Date(iso);
   224→  if (isNaN(date.getTime())) return 0;
   225→  return Math.max(0, Math.floor((Date.now() - date.getTime()) / 1000));
   226→}

  1087→  // ----- Telegram state -----
  1088→  const [telegram, setTelegram] = React.useState<TelegramResponse | null>(null);
  1089→  const [tgLoading, setTgLoading] = React.useState(false);
  1090→  const [tgRefreshing, setTgRefreshing] = React.useState(false);
  1091→  const [tgError, setTgError] = React.useState<string | null>(null);
  1092→  const [autoRefresh, setAutoRefresh] = React.useState(false);
  1093→
  1094→  // ----- Sources state -----
  1095→  const [sources, setSources] = React.useState<DataSourceInfo[]>([]);
  1096→
  1097→  // ----- Cache age display tick (re-render every 1s for smooth "Xs ago") -----
  1098→  const [, setTick] = React.useState(0);
  1099→  React.useEffect(() => {
  1100→    const id = setInterval(() => setTick((x) => x + 1), 1000);
  1101→    return () => clearInterval(id);
  1102→  }, []);
  1103→
  1104→  // ----- Fetchers -----
  1105→  const fetchNews = React.useCallback(async (refresh: boolean) => {

  1107→    else setNewsLoading(true);
  1108→    setNewsError(null);

  1110→      const res = await fetch("/api/scanner/news?limit=40");
  1111→      if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
  1112→      const data = (await res.json()) as NewsResponse;
  1113→      if (!data || !Array.isArray(data.articles)) {
  1114→        throw new Error("Invalid response shape");
  1115→      }
  1116→      setNews(data);
  1117→    } catch (e) {
  1118→      setNewsError(e instanceof Error ? e.message : "Failed to load news");
  1119→    } finally {
  1120→      setNewsLoading(false);
  1121→      setNewsRefreshing(false);
  1122→    }
  1123→  }, []);
  1124→
  1125→  const fetchNewsFa = React.useCallback(async (refresh: boolean) => {
  1126→    if (refresh) setNewsFaRefreshing(true);
  1127→    else setNewsFaLoading(true);
  1128→    setNewsFaError(null);
  1129→    try {
  1130→      const res = await fetch("/api/scanner/news/fa?limit=40");
  1131→      if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
  1132→      const data = (await res.json()) as PersianNewsResponse;
  1133→      if (!data || !Array.isArray(data.articles)) {
  1134→        throw new Error("Invalid response shape");
  1135→      }
  1136→      setNewsFa(data);
  1137→    } catch (e) {
  1138→      setNewsFaError(e instanceof Error ? e.message : "Failed to load Persian news");
  1139→    } finally {
  1140→      setNewsFaLoading(false);
  1141→      setNewsFaRefreshing(false);
  1142→    }
  1143→  }, []);
  1144→
  1145→  const fetchTelegram = React.useCallback(async (refresh: boolean) => {
  1146→    if (refresh) setTgRefreshing(true);
  1147→    else setTgLoading(true);
  1148→    setTgError(null);
  1149→    try {
  1150→      const res = await fetch(
  1151→        "/api/scanner/telegram?channel=Mastersharkcrypto&limit=20",
  1152→      );
  1153→      if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
  1154→      const data = (await res.json()) as TelegramResponse;
  1155→      if (data?.error) throw new Error(data.error);
  1156→      if (!data || !Array.isArray(data.messages)) {
  1157→        throw new Error("Invalid response shape");
  1158→      }
  1159→      setTelegram(data);
  1160→    } catch (e) {
  1161→      setTgError(e instanceof Error ? e.message : "Failed to load telegram feed");
  1162→    } finally {
  1163→      setTgLoading(false);
  1164→      setTgRefreshing(false);
  1165→    }
  1166→  }, []);
  1167→
  1168→  const fetchSources = React.useCallback(async () => {
  1169→    try {
  1170→      const res = await fetch("/api/scanner/sources");
  1171→      if (!res.ok) return;
  1172→      const data = (await res.json()) as SourcesStatus;
  1173→      if (data?.sources) setSources(data.sources);
  1174→    } catch {
  1175→      // Non-critical — silently ignore.
  1176→    }
  1177→  }, []);
  1178→
  1179→  // ----- Initial load (sources + active tab) -----
  1180→  React.useEffect(() => {
  1181→    fetchSources();
  1182→  }, [fetchSources]);
  1183→
  1184→  React.useEffect(() => {
  1185→    if (tab === "news" && !news && !newsLoading) fetchNews(false);
  1186→    else if (tab === "news_fa" && !newsFa && !newsFaLoading) fetchNewsFa(false);
  1187→    else if (tab === "telegram" && !telegram && !tgLoading) fetchTelegram(false);
  1188→  }, [tab, news, newsFa, telegram, newsLoading, newsFaLoading, tgLoading, fetchNews, fetchNewsFa, fetchTelegram]);
  1189→
  1190→  // ----- Auto-refresh telegram every 60s when toggle is on -----
  1191→  React.useEffect(() => {
  1192→    if (!autoRefresh) return;
  1193→    const id = setInterval(() => fetchTelegram(true), 60_000);
  1194→    return () => clearInterval(id);
  1195→  }, [autoRefresh, fetchTelegram]);
  1196→
  1197→  // ----- Derived: news sources list for filter chips -----
  1198→  const newsSources = React.useMemo(() => {
  1199→    const set = new Set<string>();
  1200→    news?.articles.forEach((a) => set.add(a.source));
  1201→    return Array.from(set);
  1202→  }, [news]);
  1203→
  1204→  // ----- Derived: filtered news articles (English) -----
  1205→  const filteredArticles = React.useMemo(() => {
  1206→    if (!news) return [];
  1207→    const q = search.trim().toLowerCase();
  1208→    return news.articles.filter((a) => {
  1209→      if (sourceFilter !== "all" && a.source !== sourceFilter) return false;
  1210→      if (q) {
  1211→        const hay = `${a.title} ${a.summary}`.toLowerCase();
  1212→        if (!hay.includes(q)) return false;
  1213→      }
  1214→      return true;
  1215→    });
  1216→  }, [news, sourceFilter, search]);
  1217→
  1218→  // ----- Derived: filtered Persian news articles -----
  1219→  const filteredFaArticles = React.useMemo(() => {
  1220→    if (!newsFa) return [];
  1221→    const q = searchFa.trim().toLowerCase();
  1222→    return newsFa.articles.filter((a) => {
  1223→      if (categoryFilter !== "all" && a.category !== categoryFilter) return false;
  1224→      if (q) {
  1225→        const hay = `${a.title} ${a.summary}`.toLowerCase();
  1226→        if (!hay.includes(q)) return false;
  1227→      }
  1228→      return true;
  1229→    });
  1230→  }, [newsFa, categoryFilter, searchFa]);
  1231→
  1232→  // ----- Derived: cache age for the active tab -----
  1233→  const activeFetchedAt =
  1234→    tab === "news" ? news?.fetched_at
  1235→    : tab === "news_fa" ? newsFa?.fetched_at
  1236→    : telegram?.fetched_at;
  1237→  // secondsSince recomputes on every render (tick state forces re-render every 1s).
  1238→  void secondsSince(activeFetchedAt ?? null);
  1239→
  1240→  // ----- Refresh handler for the current tab -----
  1241→  const handleRefresh = React.useCallback(() => {
  1242→    if (tab === "news") fetchNews(true);
  1243→    else if (tab === "news_fa") fetchNewsFa(true);
  1244→    else fetchTelegram(true);
  1245→  }, [tab, fetchNews, fetchNewsFa, fetchTelegram]);
  1246→
  1247→  const isRefreshing =
  1248→    tab === "news" ? newsRefreshing
  1249→    : tab === "news_fa" ? newsFaRefreshing
  1250→    : tgRefreshing;
  1251→  const isLoading =
  1252→    tab === "news" ? newsLoading
  1253→    : tab === "news_fa" ? newsFaLoading
  1254→    : tgLoading;
  1255→  const activeError =
  1256→    tab === "news" ? newsError
  1257→    : tab === "news_fa" ? newsFaError
  1258→    : tgError;
  1259→  const onRetry =
  1260→    tab === "news" ? () => fetchNews(false)
  1261→    : tab === "news_fa" ? () => fetchNewsFa(false)
  1262→    : () => fetchTelegram(false);
  1263→
  1264→  // ----------------------------------------------------------------------- //
  1265→  //  Render
  1266→  // ----------------------------------------------------------------------- //
  1267→
  1268→  return (
  1269→    <TooltipProvider>
  1270→      <div className="space-y-4">
  1271→        {/* ===== Header ===== */}
  1272→        <Card className="border-border/60 bg-card/40 backdrop-blur-sm">
  1273→          <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
  1274→            <div className="flex items-center gap-3">
  1275→              <div className="flex size-10 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500/20 to-teal-500/20 text-emerald-600 dark:text-emerald-400">
  1276→                <Newspaper className="size-5" />
  1277→              </div>
  1278→              <div>
  1279→                <h2 className="text-lg font-semibold leading-tight">
  1280→                  {tt("news.title", "News & Signals")}
  1281→                </h2>
  1282→                <p className="text-xs text-muted-foreground">
  1283→                  {tt("news.subtitle", "Crypto news + Telegram channel, unified.")}
  1284→                </p>
  1285→              </div>
  1286→            </div>
  1287→            <div className="flex items-center gap-3">
  1288→              {activeFetchedAt && (
  1289→                <span className="hidden items-center gap-1 text-xs text-muted-foreground sm:flex">
  1290→                  <Clock className="size-3" />
  1291→                  {tt("news.updated", "Updated")} {timeAgo(activeFetchedAt)}
  1292→                </span>
  1293→              )}
  1294→              <Button
  1295→                size="sm"
  1296→                variant="outline"
  1297→                onClick={handleRefresh}
  1298→                disabled={isRefreshing || isLoading}
  1299→              >
  1300→                <RefreshCw className={`size-3.5 ${isRefreshing ? "animate-spin" : ""}`} />
  1301→                <span className="hidden sm:inline">{tt("news.refresh", "Refresh")}</span>
  1302→              </Button>
  1303→            </div>
  1304→          </CardContent>
  1305→        </Card>
  1306→
  1307→        {/* ===== Data sources badge row ===== */}
  1308→        <SourcesBadgeRow sources={sources} />
  1309→
  1310→        {/* ===== Tabs ===== */}
  1311→        <Tabs
  1312→          value={tab}
  1313→          onValueChange={(v) => setTab(v as "news" | "news_fa" | "telegram")}
  1314→          className="w-full"
  1315→        >
  1316→          <TabsList className="w-full sm:w-auto">
  1317→            <TabsTrigger value="news" className="flex-1 sm:flex-initial">
  1318→              <Newspaper className="size-4" />
  1319→              {tt("news.tab", "Crypto News")}
  1320→              {news && news.articles.length > 0 && (
  1321→                <Badge variant="secondary" className="ml-1 px-1.5 py-0 text-[10px]">
  1322→                  {news.articles.length}
  1323→                </Badge>
  1324→              )}
  1325→            </TabsTrigger>
  1326→            <TabsTrigger value="news_fa" className="flex-1 sm:flex-initial">
  1327→              <BookOpen className="size-4" />
  1328→              {tt("news_fa.tab", "اخبار فارسی")}

  1330→                <Badge variant="secondary" className="ml-1 px-1.5 py-0 text-[10px]">
  1331→                  {newsFa.articles.length}
  1332→                </Badge>
  1333→              )}
  1334→            </TabsTrigger>
  1335→            <TabsTrigger value="telegram" className="flex-1 sm:flex-initial">
  1336→              <Send className="size-4" />
  1337→              {tt("telegram.tab", "Telegram Channel")}
  1338→              {telegram && telegram.messages.length > 0 && (

  1340→                  {telegram.messages.length}

  1369→              categoryFilter={categoryFilter}
