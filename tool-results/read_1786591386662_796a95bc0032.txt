   229→
   230→    # thesis (i18n)

   233→    # institutional adoption (i18n)

   329→#  Investment analysis
   330→# --------------------------------------------------------------------------- #
   331→def _build_investment_analysis(
   332→    ev: EvidenceBundle, project_quality: float, token_q: float | None, confidence: float
   333→) -> InvestmentAnalysis:
   334→    mc = ev.market_cap_usd
   335→    fdv = ev.fdv_usd or mc
   336→    annual_rev = (ev.economic.revenue or 0.0) * 365.0
   337→    annual_fees = (ev.economic.fees or 0.0) * 365.0
   338→    tvl = ev.economic.tvl or 0.0
   339→
   340→    # Legacy multiples (kept for backward compat in report)
   341→    fdv_rev = (fdv / annual_rev) if (fdv and annual_rev > 0) else None
   342→    mc_rev = (mc / annual_rev) if (mc and annual_rev > 0) else None
   343→    fdv_fees = (fdv / annual_fees) if (fdv and annual_fees > 0) else None
   344→    mc_tvl = (mc / tvl) if (mc and tvl > 0) else None
   345→
   346→    # Framework 3.0: P/R, P/F, P/T valuation multiples
   347→    # P/R = MC / Annualized Revenue (real revenue only)
   348→    # P/F = FDV / Annualized Fees (fees ≠ revenue, explicitly)
   349→    # P/T = MC / TVL
   350→    p_r = (mc / annual_rev) if (mc and annual_rev > 0) else None
   351→    p_f = (fdv / annual_fees) if (fdv and annual_fees > 0) else None
   352→    p_t = (mc / tvl) if (mc and tvl > 0) else None
   353→
   354→    # Fee volatility check — uses REAL 7d and 30d fees from DeFiLlama
   355→    # Framework 3.0: "If Fee Volatility > 40%, P/R Reliability Low"
   356→    f7d_total = ev.economic.fees_7d or 0.0   # total 7d fees
   357→    f30d_total = ev.economic.fees_30d or 0.0  # total 30d fees
   358→    f24h = ev.economic.fees or 0.0            # 24h fees
   359→    fee_vol = None
   360→    if f7d_total > 0 and f30d_total > 0:
   361→        # Compare daily averages: (7d avg daily) vs (30d avg daily)
   362→        avg_7d_daily = f7d_total / 7.0
   363→        avg_30d_daily = f30d_total / 30.0
   364→        if avg_30d_daily > 0:
   365→            fee_vol = abs(avg_7d_daily - avg_30d_daily) / avg_30d_daily * 100
   366→    elif f24h > 0 and f7d_total > 0:
   367→        # Fallback: compare 24h vs 7d average
   368→        avg_7d_daily = f7d_total / 7.0
   369→        if avg_7d_daily > 0:
   370→            fee_vol = abs(f24h - avg_7d_daily) / avg_7d_daily * 100
   371→
   372→    # Fee stability label
   373→    fee_stability = "unknown"
   374→    if fee_vol is not None:
   375→        if fee_vol > 40:
   376→            fee_stability = "volatile"
   377→        else:
   378→            fee_stability = "stable"
   379→
   380→    # Valuation verdict: Framework 3.0 prefers P/R, falls back to P/F, then P/T
   381→    valuation = ValuationVerdict.UNABLE
   382→    if p_r is not None:
   383→        if p_r < 8:
   384→            valuation = ValuationVerdict.CHEAP
   385→        elif p_r < 20:
   386→            valuation = ValuationVerdict.FAIR
   387→        else:
   388→            valuation = ValuationVerdict.EXPENSIVE
   389→    elif p_f is not None:
   390→        # Using P/F as proxy (fees ≠ revenue, so wider thresholds)
   391→        if p_f < 10:
   392→            valuation = ValuationVerdict.CHEAP
   393→        elif p_f < 25:
   394→            valuation = ValuationVerdict.FAIR
   395→        else:
   396→            valuation = ValuationVerdict.EXPENSIVE
   397→    elif p_t is not None:
   398→        if p_t < 0.15:
   399→            valuation = ValuationVerdict.CHEAP
   400→        elif p_t < 0.4:
   401→            valuation = ValuationVerdict.FAIR
   402→        else:
   403→            valuation = ValuationVerdict.EXPENSIVE
   404→
   405→    # valuation attractiveness (0-100): lower multiples = more attractive
   406→    val_attr = None
   407→    if p_r is not None:
   408→        val_attr = max(0.0, min(100.0, 100.0 - p_r * 2.5))
   409→    elif p_f is not None:
   410→        val_attr = max(0.0, min(100.0, 100.0 - p_f * 2.0))
   411→    elif p_t is not None:
   412→        val_attr = max(0.0, min(100.0, 100.0 - p_t * 150.0))
   413→
   414→    # timing: blend momentum (24h change) + cycle position
   415→    timing = 50.0
   416→    if ev.economic.revenue_growth_pct is not None:
   417→        timing = max(0.0, min(100.0, 50.0 + ev.economic.revenue_growth_pct * 0.5))
   418→
   419→    # investment attractiveness = blend of project, token, valuation, timing
   420→    inv_attr = None
   421→    if val_attr is not None:
   422→        inv_attr = 0.35 * project_quality + 0.25 * (token_q or 50) + 0.25 * val_attr + 0.15 * timing
   423→        inv_attr = round(max(0.0, min(100.0, inv_attr)), 1)
   424→
   425→    return InvestmentAnalysis(
   426→        fdv_revenue=round(fdv_rev, 2) if fdv_rev else None,
   427→        mc_revenue=round(mc_rev, 2) if mc_rev else None,
   428→        fdv_fees=round(fdv_fees, 2) if fdv_fees else None,
   429→        mc_fees=round(mc / annual_fees, 2) if (mc and annual_fees > 0) else None,
   430→        mc_tvl=round(mc_tvl, 2) if mc_tvl else None,
   431→        p_r=round(p_r, 2) if p_r is not None else None,
   432→        p_f=round(p_f, 2) if p_f is not None else None,
   433→        p_t=round(p_t, 2) if p_t is not None else None,
   434→        annualized_fees=round(annual_fees, 0) if annual_fees > 0 else None,
   435→        annualized_revenue=round(annual_rev, 0) if annual_rev > 0 else None,
   436→        fee_volatility_pct=round(fee_vol, 1) if fee_vol is not None else None,
   437→        valuation=valuation,
   438→        project_quality=round(project_quality, 1),
   439→        token_quality=token_q,
   440→        valuation_attractiveness=round(val_attr, 1) if val_attr is not None else None,
   441→        timing=round(timing, 1),
   442→        investment_attractiveness=inv_attr,
   443→        cycle_phase=CyclePhase.HIDDEN_DEV,  # set later
   444→        catalysts=[],
   445→        thesis="",
   446→        kill_conditions=[],
   447→    )

   450→def _infer_cycle_phase(ev: EvidenceBundle, project_quality: float) -> CyclePhase:
   451→    rev = ev.economic.revenue or 0.0
   452→    tvl = ev.economic.tvl or 0.0
   453→    if ev.has_regulatory_license and rev * 365 > 50_000_000:
   454→        return CyclePhase.INSTITUTIONAL
   455→    if rev * 365 > 100_000_000:
   456→        return CyclePhase.SCALE
   457→    if rev * 365 > 10_000_000:
   458→        return CyclePhase.MONETIZATION
   459→    if tvl > 500_000_000:
   460→        return CyclePhase.ADOPTION
   461→    if tvl > 50_000_000:
   462→        return CyclePhase.PMF
   463→    return CyclePhase.HIDDEN_DEV

   466→def _peer_percentile(score: float) -> float:
   467→    # very rough mapping of absolute score → percentile
   468→    if score >= 90:
   469→        return 98.0
   470→    if score >= 80:
   471→        return 90.0
   472→    if score >= 70:
   473→        return 75.0
   474→    if score >= 60:
   475→        return 55.0
   476→    if score >= 50:
   477→        return 35.0
   478→    if score >= 40:
   479→        return 20.0
   480→    return 8.0
   481→
   482→
   483→def _rank_from_score(score: float) -> int:
   484→    if score >= 90:
   485→        return 1
   486→    if score >= 80:
   487→        return 3
   488→    if score >= 70:
   489→        return 8
   490→    if score >= 60:
   491→        return 20
   492→    if score >= 50:
   493→        return 50
   494→    return 100

   497→def _comparables(category: str, self_name: str = "") -> list[str]:
   498→    c = (category or "").lower()
   499→    sn = (self_name or "").lower()
   500→    pool: list[str]
