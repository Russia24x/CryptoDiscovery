   111→
   112→/** `$1.27T` · `$1.2B` · `$340M` · `$12K` · `$0.034` · `—` for null. */
   113→function fmtUsd(v: number | null | undefined): string {
   114→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   115→  const abs = Math.abs(v);
   116→  const sign = v < 0 ? "-" : "";
   117→  if (abs >= 1e12) return `${sign}$${(abs / 1e12).toFixed(2)}T`;
   118→  if (abs >= 1e9) return `${sign}$${(abs / 1e9).toFixed(2)}B`;
   119→  if (abs >= 1e6) return `${sign}$${(abs / 1e6).toFixed(1)}M`;
   120→  if (abs >= 1e3) return `${sign}$${(abs / 1e3).toFixed(1)}K`;
   121→  if (abs >= 1) return `${sign}$${abs.toFixed(2)}`;
   122→  if (abs > 0) return `${sign}$${abs.toFixed(4)}`;
   123→  return "$0.00";
   124→}
   125→
   126→/** Like fmtUsd but tuned for token prices (more decimals for sub-$1 tokens). */
   127→function fmtPrice(v: number | null | undefined): string {
   128→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   129→  const abs = Math.abs(v);
   130→  if (abs >= 1e3) return `$${v.toLocaleString(undefined, { maximumFractionDigits: 0 })}`;
   131→  if (abs >= 1) return `$${v.toFixed(2)}`;
   132→  if (abs >= 0.01) return `$${v.toFixed(4)}`;
   133→  if (abs > 0) return `$${v.toExponential(3)}`;
   134→  return "$0.00";
   135→}
   136→
   137→/** `+1.2%` / `-2.1%` / `—`. */
   138→function fmtPct(v: number | null | undefined, digits = 2): string {
   139→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   140→  const s = v > 0 ? "+" : "";
   141→  return `${s}${v.toFixed(digits)}%`;
   142→}
   143→
   144→/** `18,404` with commas. */
   145→function fmtNum(v: number | null | undefined): string {
   146→  if (v == null || typeof v !== "number" || isNaN(v)) return "—";
   147→  return v.toLocaleString();
   148→}
   149→
   150→/** Color class for a % change: emerald for positive, rose for negative. */
   151→function pctColor(v: number | null | undefined): string {
   152→  if (v == null || isNaN(v)) return "text-muted-foreground";
   153→  if (v > 0) return "text-emerald-500";
   154→  if (v < 0) return "text-rose-500";
   155→  return "text-muted-foreground";
   156→}

   211→function useMarketOverview() {
   212→  const [data, setData] = React.useState<MarketOverview | null>(null);
   213→  const [loading, setLoading] = React.useState(true);
   214→  const [refreshing, setRefreshing] = React.useState(false);
   215→  const [error, setError] = React.useState<string | null>(null);
   216→
   217→  const load = React.useCallback(async (isRefresh: boolean) => {
   218→    if (isRefresh) setRefreshing(true);
   219→    else setLoading(true);
   220→    setError(null);
   221→    try {
   222→      const ctrl = new AbortController();
   223→      const t = setTimeout(() => ctrl.abort(), 60_000);
   224→      const res = await fetch("/api/scanner/market/overview", {
   225→        signal: ctrl.signal,
   226→        cache: "no-store",
   227→      });
   228→      clearTimeout(t);
   229→      if (!res.ok) {
   230→        throw new Error(`HTTP ${res.status}`);
   231→      }
   232→      const json: MarketOverview = await res.json();
   233→      setData(json);
   234→    } catch (err) {
   235→      const msg = err instanceof Error ? err.message : String(err);
   236→      if (err instanceof DOMException && err.name === "AbortError") {
   237→        setError("Request timed out after 60s.");
   238→      } else {
   239→        setError(msg);
   240→      }
   241→    } finally {
   242→      setLoading(false);
   243→      setRefreshing(false);
   244→    }
   245→  }, []);
   246→
   247→  React.useEffect(() => {
   248→    load(false);
   249→  }, [load]);
   250→
   251→  return { data, loading, refreshing, error, refresh: () => load(true) };
   252→}
   253→
   254→/** Returns the age in seconds of the cached_at timestamp, re-ticking every 10s. */
   255→function useCacheAge(cachedAt: string | null | undefined): number | null {
   256→  const [now, setNow] = React.useState(() => Date.now());
   257→  React.useEffect(() => {
   258→    const id = setInterval(() => setNow(Date.now()), 10_000);
   259→    return () => clearInterval(id);
   260→  }, []);
   261→  if (!cachedAt) return null;
   262→  const t = Date.parse(cachedAt);
   263→  if (isNaN(t)) return null;
   264→  return Math.max(0, Math.floor((now - t) / 1000));
   265→}
   266→
   267→function formatAge(secs: number | null): string {
   268→  if (secs == null) return "";
   269→  if (secs < 60) return `${secs}s ago`;
   270→  const m = Math.floor(secs / 60);
   271→  const s = secs % 60;
   272→  if (m < 60) return `${m}m ${s}s ago`;
   273→  const h = Math.floor(m / 60);
   274→  return `${h}h ${m % 60}m ago`;
   275→}

   712→const SECTOR_COLORS = [
   713→  "#10b981", // emerald

   716→  "#f59e0b", // amber

   724→
   725→function SectorsChart({ sectors }: { sectors: SectorBreakdown[] }) {
   726→  const sorted = React.useMemo(
   727→    () => [...sectors].sort((a, b) => (b.total_market_cap || 0) - (a.total_market_cap || 0)),
   728→    [sectors],
   729→  );
   730→  const total = React.useMemo(
   731→    () => sorted.reduce((sum, s) => sum + (s.total_market_cap || 0), 0),
   732→    [sorted],
   733→  );
   734→  const [hovered, setHovered] = React.useState<number | null>(null);
   735→
   736→  if (sorted.length === 0) {
   737→    return (
   738→      <div className="flex h-32 items-center justify-center text-sm text-muted-foreground">
   739→        No sector data
   740→      </div>
   741→    );
   742→  }
   743→
   744→  return (
   745→    <div className="space-y-4">
   746→      {/* Proportional stacked bar */}
   747→      <div className="flex h-6 w-full overflow-hidden rounded-md border border-border/40">
   748→        {sorted.map((s, i) => {
   749→          const v = s.total_market_cap || 0;
   750→          const w = total > 0 ? (v / total) * 100 : 0;
   751→          if (w <= 0) return null;
   752→          return (
   753→            <Tooltip key={s.sector}>
   754→              <TooltipTrigger asChild>
   755→                <div
   756→                  className="h-full transition-all"
   757→                  style={{
   758→                    width: `${w}%`,
   759→                    backgroundColor: SECTOR_COLORS[i % SECTOR_COLORS.length],
   760→                    opacity: hovered == null || hovered === i ? 1 : 0.35,
   761→                  }}
   762→                  onMouseEnter={() => setHovered(i)}
   763→                  onMouseLeave={() => setHovered(null)}
   764→                />
   765→              </TooltipTrigger>
   766→              <TooltipContent>
   767→                <div className="text-xs">
   768→                  <div className="font-medium">{s.sector}</div>
   769→                  <div className="text-muted-foreground">{fmtUsd(v)} · {((v / total) * 100).toFixed(1)}%</div>
   770→                </div>
   771→              </TooltipContent>
   772→            </Tooltip>
   773→          );
   774→        })}
   775→      </div>
   776→
   777→      {/* Per-sector horizontal bars */}
   778→      <div className="space-y-2">
   779→        {sorted.map((s, i) => {
   780→          const v = s.total_market_cap || 0;
   781→          const w = total > 0 ? (v / total) * 100 : 0;
   782→          const color = SECTOR_COLORS[i % SECTOR_COLORS.length];
   783→          return (
   784→            <div
   785→              key={s.sector}
   786→              className="grid grid-cols-[minmax(120px,1fr)_2fr_auto] items-center gap-3 text-xs"
   787→              onMouseEnter={() => setHovered(i)}
   788→              onMouseLeave={() => setHovered(null)}
   789→            >
   790→              <div className="flex items-center gap-2 truncate">
   791→                <span
   792→                  className="size-2.5 shrink-0 rounded-sm"
   793→                  style={{ backgroundColor: color }}
   794→                />
   795→                <span className="truncate text-foreground">{s.sector}</span>
   796→              </div>
   797→              <div className="relative h-2.5 overflow-hidden rounded-full bg-muted/40">
   798→                <div
   799→                  className="h-full rounded-full transition-all"
   800→                  style={{
   801→                    width: `${w}%`,
   802→                    backgroundColor: color,
   803→                    opacity: hovered == null || hovered === i ? 1 : 0.45,
   804→                  }}
   805→                />
   806→              </div>
   807→              <div className="flex items-center gap-2 tabular-nums">
   808→                <span className="font-medium text-foreground">{fmtUsd(v)}</span>
   809→                <span className="text-muted-foreground text-[10px]">
   810→                  · {s.count} coins · {w.toFixed(1)}%
   811→                </span>
   812→              </div>
   813→            </div>
   814→          );
   815→        })}
   816→      </div>
   817→    </div>
   818→  );
   819→}
