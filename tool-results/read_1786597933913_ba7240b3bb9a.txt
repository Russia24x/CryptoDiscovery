   856→// --------------------------------------------------------------------------- //
   857→//  Analyzing card — full-card loading state with cycling phase messages
   858→// --------------------------------------------------------------------------- //
   859→
   860→function AnalyzingCard({
   861→  phaseIdx,
   862→  tf,
   863→}: {
   864→  phaseIdx: number;
   865→  tf: (key: string, fallback: string, vars?: Record<string, string | number>) => string;
   866→}) {
   867→  // Fake progress that asymptotically approaches 90% — gives the user a sense
   868→  // of forward motion without lying about the actual completion %.
   869→  const [progress, setProgress] = React.useState(2);
   870→  React.useEffect(() => {
   871→    const id = window.setInterval(() => {
   872→      setProgress((p) => {
   873→        if (p >= 92) return p;
   874→        // Decelerating curve: the closer we get to 92, the slower we move.
   875→        const remaining = 92 - p;
   876→        const step = Math.max(0.4, remaining * 0.06);
   877→        return Math.min(92, p + step);
   878→      });
   879→    }, 400);
   880→    return () => window.clearInterval(id);
   881→  }, []);
   882→
   883→  return (
   884→    <Card className="border-emerald-500/30 bg-card/40 backdrop-blur-sm shadow-xl shadow-emerald-500/10 overflow-hidden">
   885→      <div className="relative">
   886→        {/* Top accent bar */}
   887→        <div className="absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent via-emerald-400/80 to-transparent" />
   888→        <CardContent className="py-12 flex flex-col items-center text-center gap-6">
   889→          {/* Animated radar-like spinner */}
   890→          <div className="relative size-20">
   891→            <div className="absolute inset-0 rounded-full border-2 border-emerald-500/20" />
   892→            <div className="absolute inset-0 rounded-full border-t-2 border-emerald-400 animate-spin" style={{ animationDuration: "1.4s" }} />
   893→            <div className="absolute inset-2 rounded-full border-2 border-teal-500/10" />
   894→            <div className="absolute inset-2 rounded-full border-b-2 border-teal-400 animate-spin" style={{ animationDuration: "2s", animationDirection: "reverse" }} />
   895→            <div className="absolute inset-0 flex items-center justify-center">
   896→              <Brain className="size-7 text-emerald-400" />
   897→            </div>
   898→          </div>
   899→
   900→          <div className="space-y-2">
   901→            <h3 className="text-lg font-semibold text-foreground flex items-center justify-center gap-2">
   902→              <Loader2 className="size-4 animate-spin text-emerald-400" />
   903→              {tf("explorer.analyzingTitle", "Running framework analysis")}
   904→            </h3>
   905→            <p
   906→              key={phaseIdx}
   907→              className="text-sm text-muted-foreground transition-opacity duration-300 max-w-md"
   908→            >
   909→              {ANALYZE_PHASE_MESSAGES[phaseIdx]}
   910→            </p>
   911→          </div>
   912→
   913→          {/* Phase dots */}
   914→          <div className="flex items-center gap-1.5">
   915→            {ANALYZE_PHASE_MESSAGES.map((_, i) => (
   916→              <span
   917→                key={i}
   918→                className={cn(
   919→                  "h-1.5 rounded-full transition-all duration-300",
   920→                  i === phaseIdx
   921→                    ? "w-6 bg-emerald-400"
   922→                    : i < phaseIdx
   923→                      ? "w-1.5 bg-emerald-500/50"
   924→                      : "w-1.5 bg-muted",
   925→                )}
   926→              />
   927→            ))}
   928→          </div>
   929→
   930→          {/* Progress bar */}
   931→          <div className="w-full max-w-md space-y-1.5">
   932→            <Progress
   933→              value={progress}
   934→              className="h-1.5 bg-muted/40 [&_[data-slot=progress-indicator]]:bg-gradient-to-r [&_[data-slot=progress-indicator]]:from-emerald-500 [&_[data-slot=progress-indicator]]:to-teal-400"
   935→            />
   936→            <div className="flex items-center justify-between text-[10px] uppercase tracking-wide text-muted-foreground/70">

   939→            </div>
   940→          </div>
   941→        </CardContent>

  1253→function KeyLinkIcon({
  1254→  href,
  1255→  icon,
  1256→  label,
  1257→}: {
  1258→  href: string | null | undefined;
  1259→  icon: React.ReactNode;
  1260→  label: string;
  1261→}) {
  1262→  if (!href) return null;
  1263→  let safeHref = href;
  1264→  try {
  1265→    // Ensure protocol-relative or bare URLs become absolute & https.
  1266→    if (!/^https?:\/\//i.test(href)) safeHref = `https://${href}`;
  1267→  } catch {
  1268→    /* ignore */
  1269→  }
  1270→  return (
  1271→    <Tooltip>
  1272→      <TooltipTrigger asChild>
  1273→        <a

  1275→          target="_blank"
  1276→          rel="noopener noreferrer"
  1277→          aria-label={label}
  1278→          className="inline-flex size-8 items-center justify-center rounded-md border border-border/60 bg-input/20 text-muted-foreground transition-all hover:bg-accent/60 hover:text-foreground hover:border-border"
  1279→        >
  1280→          {icon}
  1281→        </a>
  1282→      </TooltipTrigger>
  1283→      <TooltipContent>{label}</TooltipContent>

  1285→  );
